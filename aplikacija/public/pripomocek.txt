// Life OS – pripomoček za iPhone (aplikacija Scriptable).
// Kaže današnji napredek, opravila in cilje s strežnika doma. Dotik odpre aplikacijo.
// Od zunaj potrebuje vklopljen Tailscale, doma deluje tudi prek Wi-Fi.

const NASLOVI = [
  "https://doma.TVOJ-TAILNET.ts.net:8443",
  "http://192.168.1.10:8090",
]

const OZADJE = new Color("#0e0e12")
const TEKST = new Color("#e8e8f0")
const SIVA = new Color("#787884")
const TEMNA = new Color("#46465a")
const ORANZNA = new Color("#f59e0b")
const ZELENA = new Color("#10b981")
const RDECA = new Color("#ef4444")
const BARVA_PRIORITETE = { high: RDECA, medium: ORANZNA, low: ZELENA }

async function preberi() {
  for (const naslov of NASLOVI) {
    try {
      const r = new Request(naslov + "/api/povzetek")
      r.timeoutInterval = 8
      const p = await r.loadJSON()
      p.naslov = naslov
      return p
    } catch (e) {}
  }
  return null
}

function besedilo(stack, t, velikost, barva, krepko) {
  const x = stack.addText(t)
  x.font = krepko ? Font.boldSystemFont(velikost) : Font.systemFont(velikost)
  x.textColor = barva
  x.lineLimit = 1
  return x
}

function trak(stack, delez, sirina, barva) {
  const ctx = new DrawContext()
  ctx.size = new Size(sirina, 6)
  ctx.opaque = false
  ctx.respectScreenScale = true
  ctx.setFillColor(new Color("#1e1e28"))
  const ozadje = new Path()
  ozadje.addRoundedRect(new Rect(0, 0, sirina, 6), 3, 3)
  ctx.addPath(ozadje)
  ctx.fillPath()
  if (delez > 0) {
    ctx.setFillColor(barva)
    const p = new Path()
    p.addRoundedRect(new Rect(0, 0, Math.max(6, sirina * Math.min(1, delez)), 6), 3, 3)
    ctx.addPath(p)
    ctx.fillPath()
  }
  stack.addImage(ctx.getImage())
}

function vrsticaCilja(stack, c, velikost) {
  const v = stack.addStack()
  v.centerAlignContent()
  besedilo(v, c.title, velikost, TEKST, true)
  v.addSpacer()
  besedilo(v, `${c.spodletelo}/${c.cilj}`, velikost, RDECA, true)
  let pod
  if (!c.zadnji) pod = "Še nobenega poskusa – prvi spodletljaj šteje največ."
  else if (!c.danes_poskusil) pod = `Danes še nisi poskusil · zadnjič ${c.zadnji}`
  else pod = `Poskus ${c.zadnji}` + (c.zadnja_vrednost != null && c.merilo ? ` (${c.zadnja_vrednost}/${c.merilo})` : "")
  besedilo(stack, pod, velikost - 3, c.danes_poskusil ? SIVA : ORANZNA)
}

function glava(w, p, mali) {
  const g = w.addStack()
  g.centerAlignContent()
  const levo = g.addStack()
  levo.layoutVertically()
  besedilo(levo, "DANES", 10, TEMNA, true)
  besedilo(levo, `${p.odstotek}%`, mali ? 30 : 26, p.vse && p.odstotek === 100 ? ZELENA : TEKST, true)
  g.addSpacer()
  if (!mali) {
    const desno = g.addStack()
    desno.layoutVertically()
    besedilo(desno, `${p.narejeno} od ${p.vse} opravljenih`, 12, SIVA)
    besedilo(desno, p.niz ? `niz ${p.niz} ${p.niz === 1 ? "dan" : "dni"} · spodletelo danes ${p.spodletelo_danes}` : `spodletelo danes ${p.spodletelo_danes}`, 12, p.spodletelo_danes ? ZELENA : SIVA)
  }
  w.addSpacer(4)
  trak(w, p.vse ? p.narejeno / p.vse : 0, mali ? 130 : 300, p.odstotek === 100 ? ZELENA : ORANZNA)
}

function opravila(w, p, koliko, velikost) {
  const odprta = p.opravila.filter(o => !o.done)
  if (!p.opravila.length) {
    besedilo(w, "Za danes ni opravil.", velikost, TEMNA)
    return
  }
  if (!odprta.length) {
    besedilo(w, "Vse opravljeno! 🎉", velikost, ZELENA, true)
    return
  }
  for (const o of odprta.slice(0, koliko)) {
    const v = w.addStack()
    v.centerAlignContent()
    besedilo(v, "●", velikost - 4, o.zamujeno ? RDECA : (BARVA_PRIORITETE[o.prioriteta] || ORANZNA))
    v.addSpacer(6)
    besedilo(v, o.text, velikost, o.zamujeno ? RDECA : TEKST)
  }
  if (odprta.length > koliko) besedilo(w, `+ še ${odprta.length - koliko}`, velikost - 2, SIVA)
}

async function zgradi(p) {
  const druzina = config.widgetFamily || "large"
  const w = new ListWidget()
  w.refreshAfterDate = new Date(Date.now() + 15 * 60 * 1000)

  if (druzina.startsWith("accessory")) {
    // zaklenjeni zaslon
    if (!p) { w.addText("Life OS: ni povezave"); return w }
    const odprta = p.opravila.filter(o => !o.done)
    if (druzina === "accessoryCircular") {
      const t = w.addText(`${p.odstotek}%`)
      t.font = Font.boldSystemFont(16)
      t.centerAlignText()
      return w
    }
    besedilo(w, `${p.narejeno}/${p.vse} opravil · ${p.odstotek}%`, 13, Color.white(), true)
    besedilo(w, odprta[0] ? "▸ " + odprta[0].text : "Vse opravljeno!", 12, Color.white())
    const c = p.cilji[0]
    if (druzina === "accessoryRectangular" && c) besedilo(w, `${c.title} ${c.spodletelo}/${c.cilj}`, 11, Color.white())
    return w
  }

  w.backgroundColor = OZADJE
  w.setPadding(14, 14, 12, 14)
  if (!p) {
    besedilo(w, "Life OS", 16, TEKST, true)
    w.addSpacer(6)
    besedilo(w, "Ni povezave s strežnikom.", 12, RDECA)
    besedilo(w, "Vklopi Tailscale ali se poveži na domači Wi-Fi.", 11, SIVA).lineLimit = 2
    return w
  }
  w.url = p.naslov

  if (druzina === "small") {
    glava(w, p, true)
    w.addSpacer(8)
    opravila(w, p, 3, 12)
    w.addSpacer()
    return w
  }

  glava(w, p, false)
  w.addSpacer(10)
  opravila(w, p, druzina === "medium" ? 3 : 7, druzina === "medium" ? 13 : 15)
  w.addSpacer(10)
  const cilji = p.cilji.slice(0, druzina === "medium" ? 1 : 3)
  for (const c of cilji) {
    vrsticaCilja(w, c, druzina === "medium" ? 13 : 14)
    w.addSpacer(6)
  }
  if (druzina !== "medium") {
    w.addSpacer()
    besedilo(w, p.spodbuda, 11, SIVA).lineLimit = 2
  }
  w.addSpacer()
  return w
}

const podatki = await preberi()
const pripomocek = await zgradi(podatki)
if (config.runsInWidget || config.runsInAccessoryWidget) Script.setWidget(pripomocek)
else await pripomocek.presentLarge()
Script.complete()
